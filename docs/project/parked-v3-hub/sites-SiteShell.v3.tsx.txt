import { SiteFooter, SiteHeader } from "@halation/ui/v3";
import type { ReactNode } from "react";
import { type SiteId, getSite, getSites } from "./index";
import { LegacyShell } from "./LegacyShell";
import { legalPaths, person, socials } from "./profile";

/**
 * Chrome for inner pages (design prompt v3, section 1.5): a 72 px header with the wordmark and text links, the page, and
 * a footer whose site name bleeds off the bottom edge. Photography and music carry NO reference to any other site, only
 * the one `jerryqiu.com` button (cross-site rule, 12-vision-v2.md; tested in e2e/cross-site.spec.ts). Flying is not part
 * of the v3 pass and keeps the earlier shell.
 */
export function SiteShell({
  children,
  currentId = "hub",
  links,
  current,
  overHero = false,
}: {
  children: ReactNode;
  currentId?: SiteId;
  links: { label: string; href: string }[];
  current?: string;
  quiet?: boolean;
  cinema?: boolean;
  overHero?: boolean;
}) {
  if (currentId === "flying") return <LegacyShell currentId={currentId} links={links}>{children}</LegacyShell>;
  const hub = getSite("hub");
  const self = getSite(currentId);
  const year = new Date().getFullYear();
  const note = `© ${year} ${person.name}. All photographs and writing are © ${person.name} unless stated.`;
  const word = currentId === "hub" ? "JERRY" : currentId === "photos" ? "PHOTO" : "MUSIC";
  return (
    <>
      <a className="hl-skip" href="#main">
        Skip to content
      </a>
      <SiteHeader name={self.name} links={links} current={current} overHero={overHero} />
      <main id="main" className="hl3-main">
        {children}
      </main>
      {currentId === "hub" ? (
        <SiteFooter
          word={word}
          note={note}
          columns={[
            {
              title: "Rooms",
              links: getSites()
                .filter((s) => s.id !== "hub")
                .map((s) => ({ label: s.name, href: s.url })),
            },
            { title: "Elsewhere", links: socials.map((s) => ({ label: s.label, href: s.href })) },
            { title: "Legal", links: legalPaths.map((l) => ({ label: l.label, href: l.path })) },
          ]}
        />
      ) : (
        <SiteFooter word={word} note={note} returnHref={hub.url} />
      )}
    </>
  );
}
